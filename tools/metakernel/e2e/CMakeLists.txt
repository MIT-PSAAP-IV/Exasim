cmake_minimum_required(VERSION 3.16)
project(mk_e2e C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_LINK_DEPENDS_NO_SHARED ON)
# Same model provider as the case's exasimapp (.exasim/CMakeLists.txt), so the reference is the production build.
find_package(Exasim REQUIRED COMPONENTS gpu)
if(NOT TARGET Kokkos::kokkos)
  find_package(Kokkos REQUIRED)
endif()
if(MK_CASE_KIND STREQUAL "matlab")       # kernels compiled into each executable via frontendprovider.cpp
  set(MK_MODEL_LIB "")
else()
  exasim_add_external_builtin_model(TARGET frontend_model ID 100 KERNELS "${CASE_KERNELS}" SHARED)
  set(MK_MODEL_LIB frontend_model)
endif()
file(STRINGS ${GEN}/schedules.txt MK_SCHEDULES)
function(mk_target tname src sname sched)
  add_executable(${tname} ${src})
  target_compile_definitions(${tname} PRIVATE HAVE_PARMETIS KOKKOS_DEPENDENCE USE_PROF_API=1 _HIP)
  if(NOT MK_CASE_KIND STREQUAL "matlab")   # model library (ID 100); the MATLAB case uses its in-TU frontend-generated ABI
    target_compile_definitions(${tname} PRIVATE _BUILTINLIBRARY)
  endif()
  target_include_directories(${tname} PRIVATE ${GEN}/${sname} ${GEN}/bodies ${MKDIR} ${MKDIR}/e2e)
  target_compile_options(${tname} PRIVATE -O3 -DNDEBUG -w -ffast-math -fno-unroll-loops -fno-gpu-rdc -Rpass-analysis=kernel-resource-usage)
  if(NOT sched STREQUAL "default")
    target_compile_options(${tname} PRIVATE "SHELL:-mllvm -amdgpu-sched-strategy=${sched}")
  endif()
  set_target_properties(${tname} PROPERTIES BUILD_RPATH "$ORIGIN" ENABLE_EXPORTS ON)
  target_link_libraries(${tname} PRIVATE Exasim::headers Exasim::${EXASIM_VARIANT}lib ${MK_MODEL_LIB} Kokkos::kokkos)
  if(MK_CASE_KIND STREQUAL "matlab")
    target_compile_definitions(${tname} PRIVATE MK_FRONTEND_PROVIDER)
    target_include_directories(${tname} PRIVATE ${CASE_KERNELS}/.. ${CASE_KERNELS})   # the case's own frontendprovider.cpp + kernels
  endif()
  find_library(MK_ROCTX rocprofiler-sdk-roctx HINTS /opt/rocm-7.0.0/lib /opt/rocm/lib)
  if(MK_ROCTX)
    target_link_libraries(${tname} PRIVATE ${MK_ROCTX})
  endif()
  if(EXASIM_VARIANT MATCHES "mpi")
    find_package(MPI REQUIRED)
    target_compile_definitions(${tname} PRIVATE _MPI)
    target_link_libraries(${tname} PRIVATE MPI::MPI_CXX)
    if(MK_NOROCTHRUST)
      target_compile_options(${tname} PRIVATE "SHELL:-include ${MKDIR}/e2e/mk_norocthrust.hpp")
    endif()
  endif()
endfunction()
foreach(line ${MK_SCHEDULES})
  string(REPLACE " " ";" f "${line}")
  list(GET f 0 name)
  list(GET f 1 sched)
  mk_target(mk_e2e_${name} ${MKDIR}/e2e/mk_e2e.cpp ${name} ${sched})
  mk_target(mk_app_${name} ${MKDIR}/e2e/mk_app.cpp ${name} ${sched})
  mk_target(mk_pc_${name} ${MKDIR}/e2e/mk_pc.cpp ${name} ${sched})     # preconditioner-build harness (built on request)
endforeach()
