cmake_minimum_required(VERSION 3.20)
project(metakernel CXX)
find_package(Kokkos REQUIRED)
# one executable per schedule (listed by gen.py), so each can carry its own compiler flags
file(STRINGS ${GEN}/schedules.txt MK_SCHEDULES)
foreach(line ${MK_SCHEDULES})
  string(REPLACE " " ";" f "${line}")
  list(GET f 0 name)
  list(GET f 1 sched)
  add_executable(mk_${name} mk_bench.cpp)
  target_compile_definitions(mk_${name} PRIVATE HAVE_HIP)
  target_include_directories(mk_${name} PRIVATE ${GEN}/${name} ${GEN}/bodies ${KERNELS} ${EXASIM_COMMON} ${CMAKE_SOURCE_DIR})
  target_compile_options(mk_${name} PRIVATE -O3 -Rpass-analysis=kernel-resource-usage)
  if(NOT sched STREQUAL "default")
    target_compile_options(mk_${name} PRIVATE "SHELL:-mllvm -amdgpu-sched-strategy=${sched}")
  endif()
  target_link_libraries(mk_${name} PRIVATE Kokkos::kokkos)
endforeach()
