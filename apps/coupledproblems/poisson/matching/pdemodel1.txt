function Flux(x, uq, v, w, eta, mu, t)
  output_size(f) = 2;
  f[0] = mu[0]*uq[1];
  f[1] = mu[0]*uq[2];
end

function Source(x, uq, v, w, eta, mu, t)
  output_size(s) = 1;
  s[0] = sin(x[0]*3.141592653589793)*sin(x[1]*3.141592653589793)*1.973920880217872E+1;
end

function Tdfunc(x, uq, v, w, eta, mu, t)
  output_size(m) = 1;
  m[0] = 1.0;
end

function Ubou(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(ub) = 1;
  ub[0] = sin(x[0]*3.141592653589793)*sin(x[1]*3.141592653589793);
end

function Fbou(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fb) = 1;
  fb[0] = tau[0]*uq[0]+n[0]*mu[0]*uq[1]+n[1]*mu[0]*uq[2];
end

function FbouHdg(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fb) = 2;
  fb[0] = -uhat[0];
  fb[1] = tau[0]*(uq[0]-uhat[0])+n[0]*mu[0]*uq[1]+n[1]*mu[0]*uq[2];
end

function Fint(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fi) = 1;
  fi[0] = -uhat[0];
end

function Initu(x, eta, mu)
  output_size(ui) = 1;
  ui[0] = 0.0;
end
