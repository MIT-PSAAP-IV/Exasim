  ph = PhiWilke(uq, mu, t);
  cm0 = mu[0]*uq[0]*71.39440410660612;
  cm1 = mu[0]*uq[1]*62.50234383789392;
  cm2 = mu[0]*uq[2]*33.32655693342354;
  cm3 = mu[0]*uq[3]*35.69720205330306;
  cm4 = mu[0]*uq[4]*31.25117191894696;
  csum = cm0 + cm1 + cm2 + cm3 + cm4;
  xm0 = cm0/csum;
  xm1 = cm1/csum;
  xm2 = cm2/csum;
  xm3 = cm3/csum;
  xm4 = cm4/csum;
  mm[0] = ms[0]*xm0/ph[0] + ms[1]*xm1/ph[1] + ms[2]*xm2/ph[2] + ms[3]*xm3/ph[3] + ms[4]*xm4/ph[4];
end

function KappaMix(uq, mu, t)
  output_size(km) = 1;
  ms = MuSpec(t);
  cs = CPspec(t);
  ph = PhiWilke(uq, mu, t);
  cm0 = mu[0]*uq[0]*71.39440410660612;
  cm1 = mu[0]*uq[1]*62.50234383789392;
  cm2 = mu[0]*uq[2]*33.32655693342354;
  cm3 = mu[0]*uq[3]*35.69720205330306;
  cm4 = mu[0]*uq[4]*31.25117191894696;
  csum = cm0 + cm1 + cm2 + cm3 + cm4;
  xm0 = cm0/csum;
  xm1 = cm1/csum;
  xm2 = cm2/csum;
  xm3 = cm3/csum;
  xm4 = cm4/csum;
  lam0 = ms[0]*(cs[0] + 1.25)*593.6067359633212;
  lam1 = ms[1]*(cs[1] + 1.25)*519.6739545618868;
  lam2 = ms[2]*(cs[2] + 1.25)*277.0927067702051;
  lam3 = ms[3]*(cs[3] + 1.25)*296.8033679816606;
  lam4 = ms[4]*(cs[4] + 1.25)*259.8369772809434;
  km[0] = lam0*xm0/ph[0] + lam1*xm1/ph[1] + lam2*xm2/ph[2] + lam3*xm3/ph[3] + lam4*xm4/ph[4];
end

function TGrad(uq, mu, t)
  output_size(tg) = 3;
  es = Emass(t);
  cs = CPspec(t);
  cv0 = (cs[0] - 1.0)*593.6067359633212;
  cv1 = (cs[1] - 1.0)*519.6739545618868;
  cv2 = (cs[2] - 1.0)*277.0927067702051;
  cv3 = (cs[3] - 1.0)*296.8033679816606;
  cv4 = (cs[4] - 1.0)*259.8369772809434;
  den = mu[0]*uq[0]*cv0 + mu[0]*uq[1]*cv1 + mu[0]*uq[2]*cv2 + mu[0]*uq[3]*cv3 + mu[0]*uq[4]*cv4;
  rho = uq[0] + uq[1] + uq[2] + uq[3] + uq[4];
  ir = 1.0/rho;
  uvel = uq[5]*ir;
  vvel = uq[6]*ir;
  wvel = uq[7]*ir;
  drx = -(uq[9] + uq[10] + uq[11] + uq[12] + uq[13]);
  dry = -(uq[18] + uq[19] + uq[20] + uq[21] + uq[22]);
  drz = -(uq[27] + uq[28] + uq[29] + uq[30] + uq[31]);
  dudx = (-uq[14] - drx*uvel)*ir;
  dvdx = (-uq[15] - drx*vvel)*ir;
  dwdx = (-uq[16] - drx*wvel)*ir;
  dudy = (-uq[23] - dry*uvel)*ir;
  dvdy = (-uq[24] - dry*vvel)*ir;
  dwdy = (-uq[25] - dry*wvel)*ir;
  dudz = (-uq[32] - drz*uvel)*ir;
  dvdz = (-uq[33] - drz*vvel)*ir;
  dwdz = (-uq[34] - drz*wvel)*ir;
  ke2 = 0.5*(uvel*uvel + vvel*vvel + wvel*wvel);
  dkedx = uvel*dudx + vvel*dvdx + wvel*dwdx;
  dkedy = uvel*dudy + vvel*dvdy + wvel*dwdy;
  dkedz = uvel*dudz + vvel*dvdz + wvel*dwdz;
  dredx = -ke2*drx - rho*dkedx - uq[17];
  dredy = -ke2*dry - rho*dkedy - uq[26];
  dredz = -ke2*drz - rho*dkedz - uq[35];
  dtr0 = (-es[0]/den)*mu[0]/mu[3];
  dtr1 = (-es[1]/den)*mu[0]/mu[3];
  dtr2 = (-es[2]/den)*mu[0]/mu[3];
  dtr3 = (-es[3]/den)*mu[0]/mu[3];
  dtr4 = (-es[4]/den)*mu[0]/mu[3];
  dte = (1.0/den)*mu[2]/mu[3];
  tg[0] = dtr0*(-uq[9]) + dtr1*(-uq[10]) + dtr2*(-uq[11]) + dtr3*(-uq[12]) + dtr4*(-uq[13]) + dte*dredx;
  tg[1] = dtr0*(-uq[18]) + dtr1*(-uq[19]) + dtr2*(-uq[20]) + dtr3*(-uq[21]) + dtr4*(-uq[22]) + dte*dredy;
  tg[2] = dtr0*(-uq[27]) + dtr1*(-uq[28]) + dtr2*(-uq[29]) + dtr3*(-uq[30]) + dtr4*(-uq[31]) + dte*dredz;
end

function Jdiff(uq, mu, t)
  output_size(jd) = 15;
  ds = Dspec(uq, mu, t);
  dn0 = ds[0]/(mu[1]*mu[7]);
  dn1 = ds[1]/(mu[1]*mu[7]);
  dn2 = ds[2]/(mu[1]*mu[7]);
  dn3 = ds[3]/(mu[1]*mu[7]);
  dn4 = ds[4]/(mu[1]*mu[7]);
  rho = uq[0] + uq[1] + uq[2] + uq[3] + uq[4];
  ir = 1.0/rho;
  drx = -(uq[9] + uq[10] + uq[11] + uq[12] + uq[13]);
  dry = -(uq[18] + uq[19] + uq[20] + uq[21] + uq[22]);
  drz = -(uq[27] + uq[28] + uq[29] + uq[30] + uq[31]);
  dyx0 = (-uq[9]*rho - uq[0]*drx)*ir*ir;
  dyy0 = (-uq[18]*rho - uq[0]*dry)*ir*ir;
  dyz0 = (-uq[27]*rho - uq[0]*drz)*ir*ir;
  dyx1 = (-uq[10]*rho - uq[1]*drx)*ir*ir;
  dyy1 = (-uq[19]*rho - uq[1]*dry)*ir*ir;
  dyz1 = (-uq[28]*rho - uq[1]*drz)*ir*ir;
  dyx2 = (-uq[11]*rho - uq[2]*drx)*ir*ir;
  dyy2 = (-uq[20]*rho - uq[2]*dry)*ir*ir;
  dyz2 = (-uq[29]*rho - uq[2]*drz)*ir*ir;
  dyx3 = (-uq[12]*rho - uq[3]*drx)*ir*ir;
  dyy3 = (-uq[21]*rho - uq[3]*dry)*ir*ir;
  dyz3 = (-uq[30]*rho - uq[3]*drz)*ir*ir;
  dyx4 = (-uq[13]*rho - uq[4]*drx)*ir*ir;
  dyy4 = (-uq[22]*rho - uq[4]*dry)*ir*ir;
  dyz4 = (-uq[31]*rho - uq[4]*drz)*ir*ir;
  sx = dn0*dyx0 + dn1*dyx1 + dn2*dyx2 + dn3*dyx3 + dn4*dyx4;
  sy = dn0*dyy0 + dn1*dyy1 + dn2*dyy2 + dn3*dyy3 + dn4*dyy4;
  sz = dn0*dyz0 + dn1*dyz1 + dn2*dyz2 + dn3*dyz3 + dn4*dyz4;
  jd[0] = -rho*dn0*dyx0 + uq[0]*sx;
  jd[1] = -rho*dn1*dyx1 + uq[1]*sx;
  jd[2] = -rho*dn2*dyx2 + uq[2]*sx;
  jd[3] = -rho*dn3*dyx3 + uq[3]*sx;
  jd[4] = -rho*dn4*dyx4 + uq[4]*sx;
  jd[5] = -rho*dn0*dyy0 + uq[0]*sy;
  jd[6] = -rho*dn1*dyy1 + uq[1]*sy;
  jd[7] = -rho*dn2*dyy2 + uq[2]*sy;
  jd[8] = -rho*dn3*dyy3 + uq[3]*sy;
  jd[9] = -rho*dn4*dyy4 + uq[4]*sy;
  jd[10] = -rho*dn0*dyz0 + uq[0]*sz;
  jd[11] = -rho*dn1*dyz1 + uq[1]*sz;
  jd[12] = -rho*dn2*dyz2 + uq[2]*sz;
  jd[13] = -rho*dn3*dyz3 + uq[3]*sz;
  jd[14] = -rho*dn4*dyz4 + uq[4]*sz;
end

function Stress(x, uq, mu, t)
  output_size(st) = 6;
  mm = MuMix(uq, mu, t);
  mud = mm[0]/mu[4];
  rho = uq[0] + uq[1] + uq[2] + uq[3] + uq[4];
  ir = 1.0/rho;
  uvel = uq[5]*ir;
  vvel = uq[6]*ir;
  wvel = uq[7]*ir;
  drx = -(uq[9] + uq[10] + uq[11] + uq[12] + uq[13]);
  dry = -(uq[18] + uq[19] + uq[20] + uq[21] + uq[22]);
  drz = -(uq[27] + uq[28] + uq[29] + uq[30] + uq[31]);
  dudx = (-uq[14] - drx*uvel)*ir;
  dvdx = (-uq[15] - drx*vvel)*ir;
  dwdx = (-uq[16] - drx*wvel)*ir;
  dudy = (-uq[23] - dry*uvel)*ir;
  dvdy = (-uq[24] - dry*vvel)*ir;
  dwdy = (-uq[25] - dry*wvel)*ir;
  dudz = (-uq[32] - drz*uvel)*ir;
  dvdz = (-uq[33] - drz*vvel)*ir;
  dwdz = (-uq[34] - drz*wvel)*ir;
  divu = dudx + dvdy + dwdz;
  c1 = mud/mu[10];
  c2 = (2.0/3.0)*c1;
  st[0] = c2*(2.0*dudx - dvdy - dwdz);
  st[1] = c1*(dudy + dvdx);
  st[2] = c1*(dudz + dwdx);
  st[3] = c2*(2.0*dvdy - dudx - dwdz);
  st[4] = c1*(dvdz + dwdy);
  st[5] = c2*(2.0*dwdz - dudx - dvdy);
end

function Omega(uq, mu, t)
  output_size(om) = 5;
  gs = Gspec(t);
  lgT = log(t);
  cm0 = mu[0]*uq[0]*71.39440410660612;
  cm1 = mu[0]*uq[1]*62.50234383789392;
  cm2 = mu[0]*uq[2]*33.32655693342354;
  cm3 = mu[0]*uq[3]*35.69720205330306;
  cm4 = mu[0]*uq[4]*31.25117191894696;
  pterm = 9.408090904477708 - lgT;
  gc0 = gs[0] - pterm;
  gc1 = gs[1] - pterm;
  gc2 = gs[2] - pterm;
  gc3 = gs[3] - pterm;
  gc4 = gs[4] - pterm;
  lkf0 = 37.93997377657284 + (-1.6)*lgT - 113200.0/t;
  lkf1 = 36.841361487904734 + (-1.5)*lgT - 59360.0/t;
  lkf2 = 22.33270374938051 - 75500.0/t;
  lkf3 = 15.555976732804778 + (0.42)*lgT - 42938.0/t;
  lkf4 = 15.943742263813542 - 19400.0/t;
  lkb0 = lkf0 + 2.0*gc0 - gc3;
  lkb1 = lkf1 + 2.0*gc1 - gc4;
  lkb2 = lkf2 + gc0 + gc1 - gc2;
  lkb3 = lkf3 + gc0 - gc1 + gc2 - gc3;
  lkb4 = lkf4 + gc0 - gc1 - gc2 + gc4;
  kf0 = exp(lkf0);
  kb0 = exp(lkb0);
  kf1 = exp(lkf1);
  kb1 = exp(lkb1);
  kf2 = exp(lkf2);
  kb2 = exp(lkb2);
  kf3 = exp(lkf3);
  kb3 = exp(lkb3);
  kf4 = exp(lkf4);
  kb4 = exp(lkb4);
  tb0 = cm0 + cm1 + 0.2333*(cm2 + cm3 + cm4);
  tb1 = cm0 + cm1 + 0.2*(cm2 + cm3 + cm4);
  tb2 = 22.0*(cm0 + cm1 + cm2) + cm3 + cm4;
  rr0 = (kf0*cm3 - kb0*cm0*cm0)*tb0;
  rr1 = (kf1*cm4 - kb1*cm1*cm1)*tb1;
  rr2 = (kf2*cm2 - kb2*cm0*cm1)*tb2;
  rr3 = kf3*cm1*cm3 - kb3*cm0*cm2;
  rr4 = kf4*cm1*cm2 - kb4*cm0*cm4;
  om[0] = 0.0140067*(2.0*rr0 + rr2 + rr3 + rr4);
  om[1] = 0.0159994*(2.0*rr1 + rr2 - rr3 - rr4);
  om[2] = 0.0300061*(- rr2 + rr3 - rr4);
  om[3] = 0.0280134*(- rr0 - rr3);
  om[4] = 0.0319988*(- rr1 + rr4);
end

function Flux(x, uq, v, w, eta, mu, t)
  output_size(f) = 27;
  tdim = w[0]*mu[3];
  rho = uq[0] + uq[1] + uq[2] + uq[3] + uq[4];
  ir = 1.0/rho;
  uvel = uq[5]*ir;
  vvel = uq[6]*ir;
  wvel = uq[7]*ir;
  ene = uq[8]*ir;
  av = v[0];
  pdim = tdim*8.314471468617452*(mu[0]*uq[0]*71.39440410660612 + mu[0]*uq[1]*62.50234383789392 + mu[0]*uq[2]*33.32655693342354 + mu[0]*uq[3]*35.69720205330306 + mu[0]*uq[4]*31.25117191894696);
  pre = pdim/mu[2];
  hh = ene + pre*ir;
  st = Stress(x, uq, mu, tdim);
  jd = Jdiff(uq, mu, tdim);
  hs = Hspec(tdim);
  hm0 = hs[0]*tdim*593.6067359633212/(mu[1]*mu[1]);
  hm1 = hs[1]*tdim*519.6739545618868/(mu[1]*mu[1]);
  hm2 = hs[2]*tdim*277.0927067702051/(mu[1]*mu[1]);
  hm3 = hs[3]*tdim*296.8033679816606/(mu[1]*mu[1]);
  hm4 = hs[4]*tdim*259.8369772809434/(mu[1]*mu[1]);
  tg = TGrad(uq, mu, tdim);
  km = KappaMix(uq, mu, tdim);
  kap = km[0]/mu[5];
  hjx = hm0*jd[0] + hm1*jd[1] + hm2*jd[2] + hm3*jd[3] + hm4*jd[4];
  hjy = hm0*jd[5] + hm1*jd[6] + hm2*jd[7] + hm3*jd[8] + hm4*jd[9];
  hjz = hm0*jd[10] + hm1*jd[11] + hm2*jd[12] + hm3*jd[13] + hm4*jd[14];
  edx = hjx - kap*tg[0]/(mu[10]*mu[9]*mu[8]);
  edy = hjy - kap*tg[1]/(mu[10]*mu[9]*mu[8]);
  edz = hjz - kap*tg[2]/(mu[10]*mu[9]*mu[8]);
  f[0] = uq[0]*uvel + av*uq[9] + jd[0];
  f[1] = uq[1]*uvel + av*uq[10] + jd[1];
  f[2] = uq[2]*uvel + av*uq[11] + jd[2];
  f[3] = uq[3]*uvel + av*uq[12] + jd[3];
  f[4] = uq[4]*uvel + av*uq[13] + jd[4];
  f[5] = uq[5]*uvel + pre + av*uq[14] - st[0];
  f[6] = uq[6]*uvel + av*uq[15] - st[1];
  f[7] = uq[7]*uvel + av*uq[16] - st[2];
  f[8] = uq[5]*hh + av*uq[17] - (uvel*st[0] + vvel*st[1] + wvel*st[2] - edx);
  f[9] = uq[0]*vvel + av*uq[18] + jd[5];
  f[10] = uq[1]*vvel + av*uq[19] + jd[6];
  f[11] = uq[2]*vvel + av*uq[20] + jd[7];
  f[12] = uq[3]*vvel + av*uq[21] + jd[8];
  f[13] = uq[4]*vvel + av*uq[22] + jd[9];
  f[14] = uq[5]*vvel + av*uq[23] - st[1];
  f[15] = uq[6]*vvel + pre + av*uq[24] - st[3];
  f[16] = uq[7]*vvel + av*uq[25] - st[4];
  f[17] = uq[6]*hh + av*uq[26] - (uvel*st[1] + vvel*st[3] + wvel*st[4] - edy);
  f[18] = uq[0]*wvel + av*uq[27] + jd[10];
  f[19] = uq[1]*wvel + av*uq[28] + jd[11];
  f[20] = uq[2]*wvel + av*uq[29] + jd[12];
  f[21] = uq[3]*wvel + av*uq[30] + jd[13];
  f[22] = uq[4]*wvel + av*uq[31] + jd[14];
  f[23] = uq[5]*wvel + av*uq[32] - st[2];
  f[24] = uq[6]*wvel + av*uq[33] - st[4];
  f[25] = uq[7]*wvel + pre + av*uq[34] - st[5];
  f[26] = uq[7]*hh + av*uq[35] - (uvel*st[2] + vvel*st[4] + wvel*st[5] - edz);
end

function Fint(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fi) = 6;
  tdim = w[0]*mu[3];
  pdim = tdim*8.314471468617452*(mu[0]*uq[0]*71.39440410660612 + mu[0]*uq[1]*62.50234383789392 + mu[0]*uq[2]*33.32655693342354 + mu[0]*uq[3]*35.69720205330306 + mu[0]*uq[4]*31.25117191894696);
  tg = TGrad(uq, mu, tdim);
  km = KappaMix(uq, mu, tdim);
  kap = km[0]/mu[5];
  jd = Jdiff(uq, mu, tdim);
  hs = Hspec(tdim);
  hm0 = hs[0]*tdim*593.6067359633212/(mu[1]*mu[1]);
  hm1 = hs[1]*tdim*519.6739545618868/(mu[1]*mu[1]);
  hm2 = hs[2]*tdim*277.0927067702051/(mu[1]*mu[1]);
  hm3 = hs[3]*tdim*296.8033679816606/(mu[1]*mu[1]);
  hm4 = hs[4]*tdim*259.8369772809434/(mu[1]*mu[1]);
  st = Stress(x, uq, mu, tdim);
  qdim = mu[5]*mu[3]/mu[7];
  edim = mu[0]*mu[1]*mu[1]*mu[1];
  sdim = mu[4]*mu[1]/mu[7];
  qx = -qdim*kap*tg[0];
  qy = -qdim*kap*tg[1];
  qz = -qdim*kap*tg[2];
  cx = edim*(hm0*jd[0] + hm1*jd[1] + hm2*jd[2] + hm3*jd[3] + hm4*jd[4]);
  cy = edim*(hm0*jd[5] + hm1*jd[6] + hm2*jd[7] + hm3*jd[8] + hm4*jd[9]);
  cz = edim*(hm0*jd[10] + hm1*jd[11] + hm2*jd[12] + hm3*jd[13] + hm4*jd[14]);
  txx = sdim*st[0] - pdim;
  txy = sdim*st[1];
  txz = sdim*st[2];
  tyy = sdim*st[3] - pdim;
  tyz = sdim*st[4];
  tzz = sdim*st[5] - pdim;
  fi[0] = pdim;
  fi[1] = qx*n[0] + qy*n[1] + qz*n[2];
  fi[2] = cx*n[0] + cy*n[1] + cz*n[2];
  fi[3] = txx*n[0] + txy*n[1] + txz*n[2];
  fi[4] = txy*n[0] + tyy*n[1] + tyz*n[2];
  fi[5] = txz*n[0] + tyz*n[1] + tzz*n[2];
end

function Fext(x, uq, v, w, uhat, n, uext, tau, eta, mu, t)
  output_size(fb) = 9;
  twall = uext[0];
  ew = Emass(twall);
  rew = (mu[0]*uq[0]*ew[0] + mu[0]*uq[1]*ew[1] + mu[0]*uq[2]*ew[2] + mu[0]*uq[3]*ew[3] + mu[0]*uq[4]*ew[4])/mu[2];
  pival = Expression(pi);
  jw = Jdiff(uq, mu, twall);
  sqt = sqrt(twall/(2.0*pival));
  fnw0 = jw[0]*n[0] + jw[5]*n[1] + jw[10]*n[2];
  mdc0 = eta[14]*sqt*24.36404596866705*uq[0]/mu[1];
  fb[0] = fnw0 - mdc0 + tau[0]*(uq[0] - uhat[0]);
  fnw1 = jw[1]*n[0] + jw[6]*n[1] + jw[11]*n[2];
  mdc1 = eta[15]*sqt*22.796358361849965*uq[1]/mu[1];
  fb[1] = fnw1 - mdc1 + tau[0]*(uq[1] - uhat[1]);
  fnw2 = jw[2]*n[0] + jw[7]*n[1] + jw[12]*n[2];
  mdc2 = eta[16]*sqt*16.646101849087824*uq[2]/mu[1];
  fb[2] = fnw2 - mdc2 + tau[0]*(uq[2] - uhat[2]);
  fnw3 = jw[3]*n[0] + jw[8]*n[1] + jw[13]*n[2];
  mdc3 = eta[17]*sqt*17.22798212158524*uq[3]/mu[1];
  fb[3] = fnw3 - mdc3 + tau[0]*(uq[3] - uhat[3]);
  fnw4 = jw[4]*n[0] + jw[9]*n[1] + jw[14]*n[2];
  mdc4 = eta[18]*sqt*16.119459584022767*uq[4]/mu[1];
  fb[4] = fnw4 - mdc4 + tau[0]*(uq[4] - uhat[4]);
  rvref = mu[0]*mu[1];
  fb[5] = -tau[0]*(uhat[5] + uext[1]*n[0]/rvref);
  fb[6] = -tau[0]*(uhat[6] + uext[1]*n[1]/rvref);
  fb[7] = -tau[0]*(uhat[7] + uext[1]*n[2]/rvref);
  rhox = -uext[1]*n[0]/rvref;
  rhoy = -uext[1]*n[1]/rvref;
  rhoz = -uext[1]*n[2]/rvref;
  rho = uhat[0] + uhat[1] + uhat[2] + uhat[3] + uhat[4];
  ke = 0.5*(rhox*rhox + rhoy*rhoy + rhoz*rhoz)/rho;
  fb[8] = tau[0]*(rew + ke - uhat[8]);
end

function FbouHdg(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fb) = 81;
  twall = mu[11];
  ew = WallEnergy(uq, mu);
  for i in 0:8
  fb[i] = tau[0]*(eta[i] - uhat[i]);
  endfor
  for i in 0:8
  fb[9+i] = tau[0]*(uq[i] - uhat[i]);
  endfor
  for i in 0:4
  fb[18+i] = tau[0]*(uq[i] - uhat[i]);
  endfor
  fb[23] = -tau[0]*uhat[5];
  fb[24] = -tau[0]*uhat[6];
  fb[25] = -tau[0]*uhat[7];
  fb[26] = tau[0]*(ew[0] - uhat[8]);
  mn = uq[5]*n[0] + uq[6]*n[1] + uq[7]*n[2];
  for i in 0:4
  fb[27+i] = tau[0]*(uq[i] - uhat[i]);
  endfor
  fb[32] = tau[0]*(uq[5] - mn*n[0] - uhat[5]);
  fb[33] = tau[0]*(uq[6] - mn*n[1] - uhat[6]);
  fb[34] = tau[0]*(uq[7] - mn*n[2] - uhat[7]);
  fb[35] = tau[0]*(uq[8] - uhat[8]);
  for i in 0:8
  fb[36+i] = uq[9+i]*n[0] + uq[18+i]*n[1] + uq[27+i]*n[2] + tau[0]*(uq[i] - uhat[i]);
  endfor
  cnc = w[0]*mu[3]/twall;
  renc = cnc*ew[0];
  for i in 0:4
  fb[45+i] = tau[0]*(cnc*uq[i] - uhat[i]);
  endfor
  fb[50] = -tau[0]*uhat[5];
  fb[51] = -tau[0]*uhat[6];
  fb[52] = -tau[0]*uhat[7];
  fb[53] = tau[0]*(renc - uhat[8]);
  for i in 0:4
  fb[54+i] = tau[0]*(uq[i] - uhat[i]);
  endfor
  fb[59] = -tau[0]*uhat[5];
  fb[60] = -tau[0]*uhat[6];
  fb[61] = -tau[0]*uhat[7];
  fb[62] = tau[0]*(ew[0] - uhat[8]);
  pival = Expression(pi);
  jw = Jdiff(uq, mu, twall);
  sqt = sqrt(twall/(2.0*pival));
  fnw0 = jw[0]*n[0] + jw[5]*n[1] + jw[10]*n[2];
  fnw1 = jw[1]*n[0] + jw[6]*n[1] + jw[11]*n[2];
  fnw2 = jw[2]*n[0] + jw[7]*n[1] + jw[12]*n[2];
  fnw3 = jw[3]*n[0] + jw[8]*n[1] + jw[13]*n[2];
  fnw4 = jw[4]*n[0] + jw[9]*n[1] + jw[14]*n[2];
  mdc0 = eta[14]*sqt*24.36404596866705*uq[0]/mu[1];
  mdc1 = eta[15]*sqt*22.796358361849965*uq[1]/mu[1];
  mdc2 = eta[16]*sqt*16.646101849087824*uq[2]/mu[1];
  mdc3 = eta[17]*sqt*17.22798212158524*uq[3]/mu[1];
  mdc4 = eta[18]*sqt*16.119459584022767*uq[4]/mu[1];
  fb[63] = fnw0 - mdc0 + tau[0]*(uq[0] - uhat[0]);
  fb[64] = fnw1 - mdc1 + tau[0]*(uq[1] - uhat[1]);
  fb[65] = fnw2 - mdc2 + tau[0]*(uq[2] - uhat[2]);
  fb[66] = fnw3 - mdc3 + tau[0]*(uq[3] - uhat[3]);
  fb[67] = fnw4 - mdc4 + tau[0]*(uq[4] - uhat[4]);
  fb[68] = -tau[0]*uhat[5];
  fb[69] = -tau[0]*uhat[6];
  fb[70] = -tau[0]*uhat[7];
  fb[71] = tau[0]*(ew[0] - uhat[8]);
  msn = eta[14]*sqt*24.36404596866705*uq[0]/mu[1];
  mso = eta[15]*sqt*22.796358361849965*uq[1]/mu[1];
  fb[72] = fnw0 - mdc0 - msn + tau[0]*(uq[0] - uhat[0]);
  fb[73] = fnw1 - mdc1 - mso + tau[0]*(uq[1] - uhat[1]);
  fb[74] = fnw2 - mdc2 + tau[0]*(uq[2] - uhat[2]);
  fb[75] = fnw3 - mdc3 + msn + tau[0]*(uq[3] - uhat[3]);
  fb[76] = fnw4 - mdc4 + mso + tau[0]*(uq[4] - uhat[4]);
  fb[77] = -tau[0]*uhat[5];
  fb[78] = -tau[0]*uhat[6];
  fb[79] = -tau[0]*uhat[7];
  fb[80] = tau[0]*(ew[0] - uhat[8]);
end

function EoS(x, uq, v, w, eta, mu, t)
  output_size(fe) = 1;
  pival = Expression(pi);
  ul0 = uq[0]*(atan(1.0e6*uq[0])/pival + 0.5) - atan(1.0e6)/pival + 0.5;
  ul1 = uq[1]*(atan(1.0e6*uq[1])/pival + 0.5) - atan(1.0e6)/pival + 0.5;
  ul2 = uq[2]*(atan(1.0e6*uq[2])/pival + 0.5) - atan(1.0e6)/pival + 0.5;
  ul3 = uq[3]*(atan(1.0e6*uq[3])/pival + 0.5) - atan(1.0e6)/pival + 0.5;
  ul4 = uq[4]*(atan(1.0e6*uq[4])/pival + 0.5) - atan(1.0e6)/pival + 0.5;
  tdim = w[0]*mu[3];
  hs = Hspec(tdim);
  rsum = mu[0]*ul0 + mu[0]*ul1 + mu[0]*ul2 + mu[0]*ul3 + mu[0]*ul4;
  rmx = uq[5]*mu[0]*mu[1];
  rmy = uq[6]*mu[0]*mu[1];
  rmz = uq[7]*mu[0]*mu[1];
  redim = uq[8]*mu[2] - 0.5*(rmx*rmx + rmy*rmy + rmz*rmz)/rsum;
  gsum = mu[0]*ul0*71.39440410660612*(hs[0] - 1.0) + mu[0]*ul1*62.50234383789392*(hs[1] - 1.0) + mu[0]*ul2*33.32655693342354*(hs[2] - 1.0) + mu[0]*ul3*35.69720205330306*(hs[3] - 1.0) + mu[0]*ul4*31.25117191894696*(hs[4] - 1.0);
  fe[0] = tdim*gsum - redim*0.12027222701700871;
end

function Sourcew(x, uq, v, w, eta, mu, t)
  output_size(sw) = 1;
  fe = EoS(x, uq, v, w, eta, mu, t);
  sw[0] = fe[0];
end

function Initw(x, eta, mu)
  output_size(wi) = 1;

end

function VisScalars(x, uq, v, w, eta, mu, t)
  output_size(s) = 8;
  s[0] = mu[0]*uq[0];
  s[1] = mu[0]*uq[1];
  s[2] = mu[0]*uq[2];
  s[3] = mu[0]*uq[3];
  s[4] = mu[0]*uq[4];
  s[5] = mu[0]*(uq[0] + uq[1] + uq[2] + uq[3] + uq[4]);
  s[6] = w[0]*mu[3];
  s[7] = w[0]*mu[3]*8.314471468617452*(mu[0]*uq[0]*71.39440410660612 + mu[0]*uq[1]*62.50234383789392 + mu[0]*uq[2]*33.32655693342354 + mu[0]*uq[3]*35.69720205330306 + mu[0]*uq[4]*31.25117191894696);
end

function VisVectors(x, uq, v, w, eta, mu, t)
  output_size(s) = 9;
  tdim = w[0]*mu[3];
  rhodim = mu[0]*(uq[0] + uq[1] + uq[2] + uq[3] + uq[4]);
  s[0] = uq[5]*mu[0]*mu[1]/rhodim;
  s[1] = uq[6]*mu[0]*mu[1]/rhodim;
  s[2] = uq[7]*mu[0]*mu[1]/rhodim;
  tg = TGrad(uq, mu, tdim);
  km = KappaMix(uq, mu, tdim);
  kap = km[0]/mu[5];
  s[3] = -(mu[5]*mu[3]/mu[7])*kap*tg[0];
  s[4] = -(mu[5]*mu[3]/mu[7])*kap*tg[1];
  s[5] = -(mu[5]*mu[3]/mu[7])*kap*tg[2];
  jd = Jdiff(uq, mu, tdim);
  hs = Hspec(tdim);
  hm0 = hs[0]*tdim*593.6067359633212/(mu[1]*mu[1]);
  hm1 = hs[1]*tdim*519.6739545618868/(mu[1]*mu[1]);
  hm2 = hs[2]*tdim*277.0927067702051/(mu[1]*mu[1]);
  hm3 = hs[3]*tdim*296.8033679816606/(mu[1]*mu[1]);
  hm4 = hs[4]*tdim*259.8369772809434/(mu[1]*mu[1]);
  s[6] = mu[0]*mu[1]*mu[1]*mu[1]*(hm0*jd[0] + hm1*jd[1] + hm2*jd[2] + hm3*jd[3] + hm4*jd[4]);
  s[7] = mu[0]*mu[1]*mu[1]*mu[1]*(hm0*jd[5] + hm1*jd[6] + hm2*jd[7] + hm3*jd[8] + hm4*jd[9]);
  s[8] = mu[0]*mu[1]*mu[1]*mu[1]*(hm0*jd[10] + hm1*jd[11] + hm2*jd[12] + hm3*jd[13] + hm4*jd[14]);
end

function VisTensors(x, uq, v, w, eta, mu, t)
  output_size(s) = 9;
  tdim = w[0]*mu[3];
  st = Stress(x, uq, mu, tdim);
  cdim = mu[4]*mu[1]/mu[7];
  s[0] = cdim*st[0];
  s[1] = cdim*st[1];
  s[2] = cdim*st[2];
  s[3] = cdim*st[1];
  s[4] = cdim*st[3];
  s[5] = cdim*st[4];
  s[6] = cdim*st[2];
  s[7] = cdim*st[4];
  s[8] = cdim*st[5];
end
