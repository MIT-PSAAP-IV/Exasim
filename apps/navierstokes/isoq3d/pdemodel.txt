function Flux(x, uq, v, w, eta, mu, t)
  output_size(f) = 15;

  gam = mu[0];
  gam1 = gam - 1.0;
  Re = mu[1];
  Pr = mu[2];
  Minf = mu[3];
  Tinf = mu[9];
  Tref = mu[10];
  muRef = 1.0/Re;
  M2 = Minf*Minf;
  c23 = 2.0/3.0;
  alpha = 1.0e3;
  rmin = 1.0e-2;
  pmin = 1.0e-3;
  av = v[0];

  r0 = uq[0];
  ru = uq[1];
  rv = uq[2];
  rw = uq[3];
  rE = uq[4];
  rx0 = uq[5];
  rux = uq[6];
  rvx = uq[7];
  rwx = uq[8];
  rEx = uq[9];
  ry0 = uq[10];
  ruy = uq[11];
  rvy = uq[12];
  rwy = uq[13];
  rEy = uq[14];
  rz0 = uq[15];
  ruz = uq[16];
  rvz = uq[17];
  rwz = uq[18];
  rEz = uq[19];

  rarg = r0 - rmin;
  r = rmin + rarg*(atan(alpha*rarg)/pi + 0.5) - atan(alpha)/pi + 0.5;
  dr = atan(alpha*(r-rmin))/pi + (alpha*(r-rmin))/(pi*(alpha*alpha*(r-rmin)*(r-rmin) + 1.0)) + 0.5;
  rx = rx0*dr;
  ry = ry0*dr;
  rz = rz0*dr;

  r1 = 1.0/r;
  uv = ru*r1;
  vv = rv*r1;
  wv = rw*r1;
  E = rE*r1;
  ke = 0.5*(uv*uv + vv*vv + wv*wv);
  p0 = gam1*(rE - r*ke);

  parg = p0 - pmin;
  p = pmin + parg*(atan(alpha*parg)/pi + 0.5) - atan(alpha)/pi + 0.5;
  dp = atan(alpha*(p-pmin))/pi + (alpha*(p-pmin))/(pi*(alpha*alpha*(p-pmin)*(p-pmin) + 1.0)) + 0.5;

  h = E + p*r1;

  ux = (rux - rx*uv)*r1;
  vx = (rvx - rx*vv)*r1;
  wx = (rwx - rx*wv)*r1;
  qx = uv*ux + vv*vx + wv*wx;
  px = gam1*(rEx - rx*ke - r*qx);
  gradpx = px*dp;
  Tx = (gradpx*r - p*rx)*r1*r1/gam1;

  uy = (ruy - ry*uv)*r1;
  vy = (rvy - ry*vv)*r1;
  wy = (rwy - ry*wv)*r1;
  qy = uv*uy + vv*vy + wv*wy;
  py = gam1*(rEy - ry*ke - r*qy);
  gradpy = py*dp;
  Ty = (gradpy*r - p*ry)*r1*r1/gam1;

  uz = (ruz - rz*uv)*r1;
  vz = (rvz - rz*vv)*r1;
  wz = (rwz - rz*wv)*r1;
  qz = uv*uz + vv*vz + wv*wz;
  pz = gam1*(rEz - rz*ke - r*qz);
  gradpz = pz*dp;
  Tz = (gradpz*r - p*rz)*r1*r1/gam1;

  T = p/(gam1*r);
  Tphys = Tref/Tinf*T;
  Tr = Tphys/Tref;
  muphys = muRef*sqrt(Tr*Tr*Tr)*(Tref + 110.4)/(Tphys + 110.4);
  fc = muphys*gam/Pr;

  txx = muphys*c23*(2.0*ux - vy - wz);
  txy = muphys*(uy + vx);
  txz = muphys*(uz + wx);
  tyy = muphys*c23*(2.0*vy - ux - wz);
  tyz = muphys*(vz + wy);
  tzz = muphys*c23*(2.0*wz - ux - vy);

  f[0] = ru + av*rx;
  f[1] = ru*uv + p + txx + av*rux;
  f[2] = rv*uv + txy + av*rvx;
  f[3] = rw*uv + txz + av*rwx;
  f[4] = ru*h + uv*txx + vv*txy + wv*txz + fc*Tx + av*rEx;

  f[5] = rv + av*ry;
  f[6] = ru*vv + txy + av*ruy;
  f[7] = rv*vv + p + tyy + av*rvy;
  f[8] = rw*vv + tyz + av*rwy;
  f[9] = rv*h + uv*txy + vv*tyy + wv*tyz + fc*Ty + av*rEy;

  f[10] = rw + av*rz;
  f[11] = ru*wv + txz + av*ruz;
  f[12] = rv*wv + tyz + av*rvz;
  f[13] = rw*wv + p + tzz + av*rwz;
  f[14] = rw*h + uv*txz + vv*tyz + wv*tzz + fc*Tz + av*rEz;
end

function Source(x, uq, v, w, eta, mu, t)

end

function Tdfunc(x, uq, v, w, eta, mu, t)
  output_size(m) = 5;

end

function Ubou(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(ub) = 15;

  Tinf = mu[9];
  Tref = mu[10];
  Twall = mu[11];
  TisoW = Twall/Tref*Tinf;

  ub[0] = uq[0];
  ub[1] = uq[1];
  ub[2] = uq[2];
  ub[3] = uq[3];
  ub[4] = uq[4];

  ub[5] = uq[0];
  ub[6] = 0.0;
  ub[7] = 0.0;
  ub[8] = 0.0;
  ub[9] = uq[0]*TisoW;

  ub[10] = 0.0;
  ub[11] = 0.0;
  ub[12] = 0.0;
  ub[13] = 0.0;
  ub[14] = 0.0;
end

function Fbou(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fb) = 20;

  f = Flux(x, uq, v, w, eta, mu, t);

  fi0 = f[0]*n[0] + f[5]*n[1] + f[10]*n[2] + tau[0]*(uq[0] - uhat[0]);
  fi1 = f[1]*n[0] + f[6]*n[1] + f[11]*n[2] + tau[0]*(uq[1] - uhat[1]);
  fi2 = f[2]*n[0] + f[7]*n[1] + f[12]*n[2] + tau[0]*(uq[2] - uhat[2]);
  fi3 = f[3]*n[0] + f[8]*n[1] + f[13]*n[2] + tau[0]*(uq[3] - uhat[3]);
  fi4 = f[4]*n[0] + f[9]*n[1] + f[14]*n[2] + tau[0]*(uq[4] - uhat[4]);

  fb[0] = fi0;
  fb[1] = fi1;
  fb[2] = fi2;
  fb[3] = fi3;
  fb[4] = fi4;

  fb[5] = fi0;
  fb[6] = fi1;
  fb[7] = fi2;
  fb[8] = fi3;
  fb[9] = fi4;

  fb[10] = 0.0;
  fb[11] = fi1;
  fb[12] = fi2;
  fb[13] = fi3;
  fb[14] = fi4;

  fb[15] = fi0;
  fb[16] = fi1;
  fb[17] = fi2;
  fb[18] = fi3;
  fb[19] = fi4;
end

function FbouHdg(x, uq, v, w, uhat, n, tau, eta, mu, t)
  output_size(fb) = 30;

  Tinf = mu[9];
  Tref = mu[10];
  Twall = mu[11];
  TisoW = Twall/Tref*Tinf;

  fb[0] = mu[4] - uhat[0];
  fb[1] = mu[5] - uhat[1];
  fb[2] = mu[6] - uhat[2];
  fb[3] = mu[7] - uhat[3];
  fb[4] = mu[8] - uhat[4];

  fb[5] = uq[0] - uhat[0];
  fb[6] = uq[1] - uhat[1];
  fb[7] = uq[2] - uhat[2];
  fb[8] = uq[3] - uhat[3];
  fb[9] = uq[4] - uhat[4];

  fb[10] = uq[0] - uhat[0];
  fb[11] = 0.0 - uhat[1];
  fb[12] = 0.0 - uhat[2];
  fb[13] = 0.0 - uhat[3];
  fb[14] = uhat[0]*TisoW - uhat[4];

  fb[15] = uq[5]*n[0] + uq[10]*n[1] + uq[15]*n[2] + tau[0]*(uq[0] - uhat[0]);
  fb[16] = uq[6]*n[0] + uq[11]*n[1] + uq[16]*n[2] + tau[0]*(uq[1] - uhat[1]);
  fb[17] = uq[7]*n[0] + uq[12]*n[1] + uq[17]*n[2] + tau[0]*(uq[2] - uhat[2]);
  fb[18] = uq[8]*n[0] + uq[13]*n[1] + uq[18]*n[2] + tau[0]*(uq[3] - uhat[3]);
  fb[19] = uq[9]*n[0] + uq[14]*n[1] + uq[19]*n[2] + tau[0]*(uq[4] - uhat[4]);

  f = TraceFlux(x, uq, v, w, uhat, eta, mu, t);
  fb[20] = f[0]*n[0] + f[5]*n[1] + f[10]*n[2] + tau[0]*(uq[0] - uhat[0]);
  fb[21] = f[1]*n[0] + f[6]*n[1] + f[11]*n[2] + tau[0]*(uq[1] - uhat[1]);
  fb[22] = f[2]*n[0] + f[7]*n[1] + f[12]*n[2] + tau[0]*(uq[2] - uhat[2]);
  fb[23] = f[3]*n[0] + f[8]*n[1] + f[13]*n[2] + tau[0]*(uq[3] - uhat[3]);
  fb[24] = f[4]*n[0] + f[9]*n[1] + f[14]*n[2] + tau[0]*(uq[4] - uhat[4]);

  ru = uq[1];
  rv = uq[2];
  rw = uq[3];
  run = ru*n[0] + rv*n[1] + rw*n[2];
  fb[25] = uq[0] - uhat[0];
  fb[26] = uq[1] - n[0]*run - uhat[1];
  fb[27] = uq[2] - n[1]*run - uhat[2];
  fb[28] = uq[3] - n[2]*run - uhat[3];
  fb[29] = uq[4] - uhat[4];
end

end
