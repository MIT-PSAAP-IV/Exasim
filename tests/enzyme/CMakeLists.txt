cmake_minimum_required(VERSION 3.16)
project(ExasimEnzymeFrontendTest LANGUAGES CXX)
find_package(Exasim REQUIRED COMPONENTS cpu)
if(NOT Exasim_ENZYME)
  message(FATAL_ERROR "Use an Exasim installation built with EXASIM_ENZYME=ON")
endif()
set(KERNELS "" CACHE PATH "Directory that contains the generated frontend kernels")
if(NOT IS_DIRECTORY "${KERNELS}")
  message(FATAL_ERROR "Set KERNELS to a directory with generated Kokkos sources")
endif()
exasim_add_external_builtin_model(TARGET enzyme_frontend_model ID 100 KERNELS "${KERNELS}")
add_executable(enzyme_frontend_test "../unit/enzyme_generated.cpp")
target_compile_features(enzyme_frontend_test PRIVATE cxx_std_17)
target_compile_definitions(enzyme_frontend_test PRIVATE EXASIM_EXTERNAL_PROVIDER)
target_link_libraries(enzyme_frontend_test PRIVATE
  Exasim::headers enzyme_frontend_model Kokkos::kokkos)
enable_testing()
add_test(NAME enzyme_frontend COMMAND enzyme_frontend_test 100)
