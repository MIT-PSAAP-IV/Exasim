# Focused composite-HDG contract, configuration, and CPU/HIP algebra tests.
# These tests do not submit scheduler jobs.
function(composite_hdg_unit name source)
  add_executable(${name} "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
  target_include_directories(${name} PRIVATE "${EXASIM_DIR}" "${EXASIM_DIR}/include")
  target_compile_features(${name} PRIVATE cxx_std_17)
  if(APPLE)
    # Also support this directory's standalone contract-test harness.
    target_link_options(${name} PRIVATE "LINKER:-U,_GetExasimCompositeHdgModelV1")
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${name} PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 60)
endfunction()

find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy, sympy"
    RESULT_VARIABLE _composite_python_modules OUTPUT_QUIET ERROR_QUIET)
  if(_composite_python_modules EQUAL 0)
    add_test(NAME composite_hdg_frontend_python
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/frontend_python.py"
        --converter $<TARGET_FILE:composite_hdg_config_text>)
    set_tests_properties(composite_hdg_frontend_python PROPERTIES TIMEOUT 60)
  endif()
endif()
# Interpreter presence does not establish a usable runtime/license. These optional
# language checks are enabled explicitly in the solver-layer test configuration.

if(EXASIM_COMPOSITE_LANGUAGE_TESTS AND _composite_julia)
  add_test(NAME composite_hdg_frontend_julia
    COMMAND "${_composite_julia}" --startup-file=no "${CMAKE_CURRENT_SOURCE_DIR}/frontend_julia.jl")

# Optional provider discovery uses the same GNU/Clang weak-symbol mechanism as
# the backend. These tests also cover static archive extraction, not just mocks.
if(TARGET Kokkos::kokkos AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  composite_hdg_unit(composite_hdg_finite_fastmath finite.cpp)
  target_compile_options(composite_hdg_finite_fastmath PRIVATE -O3 -ffast-math)
  target_link_libraries(composite_hdg_finite_fastmath PRIVATE Kokkos::kokkos)
  foreach(pair "contract;contract.cpp" "missing;optional.cpp")
    list(GET pair 0 name)
    list(GET pair 1 source)
    composite_hdg_unit(composite_hdg_provider_${name} "provider/${source}")
    target_link_libraries(composite_hdg_provider_${name} PRIVATE Kokkos::kokkos)
  endforeach()
  foreach(dim 2 3)
    composite_hdg_unit(composite_hdg_provider_${dim}d "provider/dimension.cpp")
    target_compile_definitions(composite_hdg_provider_${dim}d PRIVATE COMPOSITE_TEST_DIM=${dim})
    target_link_libraries(composite_hdg_provider_${dim}d PRIVATE Kokkos::kokkos)
  endforeach()
  add_library(composite_hdg_provider_archive_lib STATIC provider/archive.cpp)
  add_library(composite_hdg_provider_orphan_lib STATIC provider/adapter.cpp)
  foreach(target composite_hdg_provider_archive_lib composite_hdg_provider_orphan_lib)
    target_include_directories(${target} PRIVATE "${EXASIM_DIR}/include")
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_link_libraries(${target} PRIVATE Kokkos::kokkos)
  endforeach()
  composite_hdg_unit(composite_hdg_provider_archive provider/archive_consumer.cpp)
  target_link_libraries(composite_hdg_provider_archive PRIVATE
    composite_hdg_provider_archive_lib Kokkos::kokkos)
  composite_hdg_unit(composite_hdg_provider_orphan provider/optional.cpp)
  target_link_libraries(composite_hdg_provider_orphan PRIVATE
    composite_hdg_provider_orphan_lib Kokkos::kokkos)
  if(APPLE)
    # Mach-O resolves the weak import from the explicitly linked archive.
    # ELF leaves this member unextracted; direct/co-located adapters are portable.
    target_compile_definitions(composite_hdg_provider_orphan PRIVATE EXPECT_REGISTERED=1)
  endif()
  composite_hdg_unit(composite_hdg_provider_direct provider/optional.cpp)
  target_sources(composite_hdg_provider_direct PRIVATE provider/adapter.cpp)
  target_compile_definitions(composite_hdg_provider_direct PRIVATE EXPECT_REGISTERED=1)
  target_link_libraries(composite_hdg_provider_direct PRIVATE Kokkos::kokkos)
endif()
